cmake_minimum_required(VERSION 3.16)
project(mesh_message_lab LANGUAGES CXX)
enable_testing()
add_executable(mesh_tests tests/test.cpp)
target_include_directories(mesh_tests PRIVATE include)
set_target_properties(mesh_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(mesh_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME mesh_tests COMMAND mesh_tests)
